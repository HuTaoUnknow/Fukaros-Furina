cmake_minimum_required(VERSION 3.22)
project(Fukaros VERSION 0.4.3 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(Fukaros main.cpp mainwindow.cpp mainwindow.h mainwindow.ui settings.ui core/adb.cpp core/adb.h core/cast_source.c core/cast_source.h
 core/furina_protocol.c core/furina_protocol.h core/peer.cpp core/peer.h core/screenview.cpp core/screenview.h core/interface_support.cpp core/interface_support.h core/session_consent.cpp core/session_consent.h core/video_policy.c core/video_policy.h core/desktop_video.cpp core/desktop_video.h core/desktop_cursor.h core/cursor_shape.c core/cursor_shape.h)
target_include_directories(Fukaros PRIVATE core core/vendor/ffmpeg)
target_link_libraries(Fukaros PRIVATE Qt6::Widgets Qt6::Network)
if(MINGW)
 target_link_options(Fukaros PRIVATE -mwindows)
endif()
include(CTest)
if(BUILD_TESTING)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(protocol_test core/tests/protocol_test.c core/furina_protocol.c)
target_include_directories(protocol_test PRIVATE core core/vendor/ffmpeg)
add_test(NAME protocol COMMAND protocol_test)
add_executable(cast_source_test tests/cast_source_test.c core/cast_source.c)
target_include_directories(cast_source_test PRIVATE core)
add_test(NAME cast_source COMMAND cast_source_test)
qt_add_executable(peer_test core/tests/peer_test.cpp core/peer.cpp core/peer.h core/furina_protocol.c core/video_policy.c)
target_include_directories(peer_test PRIVATE core core/vendor/ffmpeg)
target_link_libraries(peer_test PRIVATE Qt6::Test Qt6::Network Qt6::Gui)
add_test(NAME peer COMMAND peer_test)
qt_add_executable(session_consent_test core/tests/session_consent_test.cpp core/session_consent.cpp core/session_consent.h core/peer.cpp core/peer.h core/furina_protocol.c core/video_policy.c)
target_include_directories(session_consent_test PRIVATE core)
target_link_libraries(session_consent_test PRIVATE Qt6::Test Qt6::Widgets Qt6::Network)
add_test(NAME session_consent COMMAND session_consent_test)
set_tests_properties(session_consent PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=C:/Windows/Fonts")
qt_add_executable(ui_test tests/ui_test.cpp mainwindow.cpp mainwindow.h mainwindow.ui settings.ui core/adb.cpp core/adb.h core/cast_source.c core/peer.cpp core/peer.h core/screenview.cpp core/screenview.h core/furina_protocol.c core/interface_support.cpp core/interface_support.h core/session_consent.cpp core/session_consent.h core/video_policy.c core/video_policy.h core/desktop_video.cpp core/desktop_video.h core/desktop_cursor.h core/cursor_shape.c core/cursor_shape.h)
target_include_directories(ui_test PRIVATE core core/vendor/ffmpeg ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ui_test PRIVATE Qt6::Test Qt6::Widgets Qt6::Network)
add_test(NAME ui COMMAND ui_test)
set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=C:/Windows/Fonts")
configure_file(tests/offscreen_screens.json offscreen_screens.json COPYONLY)
# Use a relative path: ':' in a Windows drive letter splits QPA platform arguments.
add_test(NAME ui_multi_screen COMMAND ui_test -platform "offscreen:configfile=offscreen_screens.json")
set_tests_properties(ui_multi_screen PROPERTIES ENVIRONMENT "QT_QPA_FONTDIR=C:/Windows/Fonts;FURINA_TEST_MULTI_SCREEN=1")
# Both UI runs own the same TCP test port and persist UI preferences.
set_tests_properties(ui ui_multi_screen PROPERTIES RUN_SERIAL TRUE)
endif()

if(WIN32)
 foreach(video_target Fukaros ui_test)
  if(TARGET ${video_target})
   target_link_libraries(${video_target} PRIVATE d3d11 dxgi ole32 winmm d3dcompiler)
  endif()
 endforeach()
endif()

if(BUILD_TESTING AND NOT ANDROID)
 add_executable(video_test core/tests/video_test.c core/video_policy.c core/furina_protocol.c)
 target_include_directories(video_test PRIVATE core)
 add_test(NAME video_policy COMMAND video_test)
endif()

foreach(display_target Fukaros ui_test)
 if(TARGET ${display_target})
  target_sources(${display_target} PRIVATE core/virtual_display.c core/virtual_display.h core/virtual_display_controller.cpp core/virtual_display_controller.h)
  if(WIN32)
   target_link_libraries(${display_target} PRIVATE user32 shell32 setupapi)
  endif()
 endif()
endforeach()
if(WIN32)
 add_executable(virtual_display_helper core/virtual_display_helper.c core/virtual_display.c)
 target_include_directories(virtual_display_helper PRIVATE core)
 target_link_libraries(virtual_display_helper PRIVATE setupapi newdev user32 advapi32 wintrust crypt32)
 if(MINGW)
  target_link_options(virtual_display_helper PRIVATE -municode -static)
 endif()
 add_custom_command(TARGET Fukaros POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/core/virtual-display/driver $<TARGET_FILE_DIR:Fukaros>/virtual-display)
 add_dependencies(Fukaros virtual_display_helper)
 if(BUILD_TESTING)
  add_executable(virtual_display_smoke core/tests/virtual_display_smoke.c core/virtual_display.c)
  target_include_directories(virtual_display_smoke PRIVATE core)
  target_link_libraries(virtual_display_smoke PRIVATE user32 setupapi)
  qt_add_executable(virtual_driver_test core/tests/virtual_driver_test.cpp)
  target_link_libraries(virtual_driver_test PRIVATE Qt6::Test)
  add_dependencies(virtual_driver_test Fukaros virtual_display_helper)
  add_test(NAME virtual_driver_package COMMAND virtual_driver_test)
 endif()
endif()

if(WIN32 AND BUILD_TESTING)
 add_executable(cursor_test core/tests/cursor_test.cpp core/cursor_shape.c)
 target_include_directories(cursor_test PRIVATE core)
 target_link_libraries(cursor_test PRIVATE d3d11 dxgi d3dcompiler)
 add_test(NAME desktop_cursor COMMAND cursor_test)
 # Manual smoke test: requires the installed driver and changes its real layout.
 qt_add_executable(display_recovery_smoke core/tests/display_recovery_smoke.cpp core/virtual_display.c core/desktop_video.cpp core/desktop_video.h core/cursor_shape.c core/video_policy.c core/furina_protocol.c core/interface_support.cpp core/interface_support.h core/screenview.cpp core/screenview.h)
 target_include_directories(display_recovery_smoke PRIVATE core core/vendor/ffmpeg)
 target_link_libraries(display_recovery_smoke PRIVATE Qt6::Widgets d3d11 dxgi ole32 winmm d3dcompiler user32 setupapi)
 # Manual hardware quality test: compares unchanged text across forced IDRs.
 # It needs an NVIDIA GPU but never changes the real display layout.
 qt_add_executable(quality_smoke core/tests/quality_smoke.cpp core/desktop_video.h core/cursor_shape.c core/video_policy.c core/furina_protocol.c core/interface_support.cpp core/interface_support.h core/screenview.cpp core/screenview.h)
 target_include_directories(quality_smoke PRIVATE core core/vendor/ffmpeg)
 target_link_libraries(quality_smoke PRIVATE Qt6::Widgets d3d11 dxgi ole32 winmm d3dcompiler)
endif()

if(WIN32)
 target_sources(Fukaros PRIVATE packaging/version.rc)
endif()

 if(TARGET ${cast_ui_target})
  target_sources(${cast_ui_target} PRIVATE core/cast_ui_state.cpp core/cast_ui_state.h core/cast_control.cpp core/cast_control.h)
